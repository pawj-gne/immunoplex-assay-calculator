import fs from 'fs'
import path from 'path'
import { app } from 'electron'

export interface AppConfig {
  serverUrl: string   // e.g. "http://192.168.1.10:3847"
  isServer: boolean   // true = this machine runs the central DB + Express server
}

/**
 * Reads config.json from userData dir. Returns null when:
 * - File does not exist → local-only mode (D-03)
 * - File is malformed JSON → local-only mode (fail-safe)
 * Must be called INSIDE app.whenReady() — app.getPath() throws before ready.
 */
export function loadConfig(): AppConfig | null {
  const configPath = path.join(app.getPath('userData'), 'config.json')
  try {
    const raw = fs.readFileSync(configPath, 'utf-8')
    return JSON.parse(raw) as AppConfig
  } catch {
    return null // missing or malformed → local-only (D-03 safe default)
  }
}
```

import type { RunRecord, RunCreate, RunUpdate } from '../../shared/types/run'
import type { Operator, OperatorCreate, OperatorUpdate } from '../../shared/types/operator'
import { localTransport } from './localTransport'

// Transport interfaces — implemented by localTransport (server/local-only mode)
// and httpTransport (client mode). IPC handlers call through these, not repositories directly.
export interface RunTransport {
  getAll(): RunRecord[]
  getById(id: string): RunRecord | null
  create(data: RunCreate): Promise<RunRecord>
  update(id: string, data: RunUpdate): Promise<RunRecord | null>
  delete(id: string): Promise<void>
}

export interface OperatorTransport {
  getAll(opts: { includeInactive: boolean }): Operator[]
  create(data: OperatorCreate): Promise<Operator>
  update(id: string, data: OperatorUpdate): Promise<Operator | null>
  softDelete(id: string): Promise<void>
}

// Module-level transport singletons — default to local (server/local-only mode)
let activeRunTransport: RunTransport = localTransport.run
let activeOperatorTransport: OperatorTransport = localTransport.operator

/**
 * Called once at startup in index.ts after DB init.
 * 'local' = server machine or local-only mode (direct repository calls)
 * 'http' = client machine (HTTP calls with offline fallback)
 */
export function initTransport(mode: 'local' | 'http', serverUrl?: string): void {
  if (mode === 'http' && serverUrl) {
    // httpTransport is imported lazily here to avoid circular dep at module scope
    // Plan 06-03 fills in httpTransport — until then, keep local as fallback
    const { httpTransport } = require('./httpTransport') as typeof import('./httpTransport')
    httpTransport.init(serverUrl)
    activeRunTransport = httpTransport.run
    activeOperatorTransport = httpTransport.operator
  }
  // 'local' leaves module vars as-is (localTransport)
}

export function getRunTransport(): RunTransport { return activeRunTransport }
export function getOperatorTransport(): OperatorTransport { return activeOperatorTransport }
```

Note: the lazy `require('./httpTransport')` in initTransport avoids a circular dependency problem — httpTransport.ts doesn't exist yet (Plan 06-03). This compiles cleanly because TypeScript sees the import type, not a runtime import at module load. When Plan 06-03 creates httpTransport.ts, this require resolves at runtime.
