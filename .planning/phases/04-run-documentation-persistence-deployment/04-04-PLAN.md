Composition (top-to-bottom per D-06):

```tsx
export function FinalizedRunView({ onStartNewRun }: { onStartNewRun: () => void }): JSX.Element {
  const currentRun = useRunStore((s) => s.runs.find((r) => r.id === s.currentRunId) ?? null)
  const plates = usePlateStore((s) => s.plates)
  const plateCount = usePlateStore((s) => s.getPlateCount())

  if (!currentRun) {
    return <div className="p-6">No active run. Save a run first or load one from Past Runs.</div>
  }

  const reqDisplay = currentRun.requestOverrideAdHoc
    ? 'Ad-hoc run'
    : `Request ${String(currentRun.requestNumber).padStart(5, '0')}`

  // Sort plate numbers ascending for consistent rendering
  const plateNumbers = Object.keys(plates).map(Number).sort((a, b) => a - b)

  return (
    <div className="space-y-6">
      {/* Header */}
      <FinalizedRunHeader />

      {/* Actions row: Print + Start New Run */}
      <div className="flex items-center gap-3 print:hidden">
        <PrintButton />
        <button
          type="button"
          onClick={() => {
            // D-10: reset all stores + back to step 1
            useCalculatorStore.getState().reset()  // existing reset cascades to plateStore + selectionStore + platformStore
            useRunStore.getState().clearCurrentRun()
            onStartNewRun()
          }}
          className="px-4 py-2 rounded bg-blue-600 text-white hover:bg-blue-700"
        >
          Start New Run
        </button>
      </div>

      {/* Prep recipe */}
      <section>
        <h3 className="text-lg font-semibold mb-2">Reagent Prep Recipe</h3>
        <PrepSheet />
      </section>

      {/* Reagent checklist */}
      <section>
        <h3 className="text-lg font-semibold mb-2">Reagent Checklist</h3>
        <ReagentChecklist />
      </section>

      {/* Bead regions */}
      <section>
        <h3 className="text-lg font-semibold mb-2">Bead Regions</h3>
        <BeadRegionList />
      </section>

      {/* Plates — one grid per plate, each with its own "Request XXXXX — Plate N of M" label */}
      <section>
        <h3 className="text-lg font-semibold mb-2">Plate Layout</h3>
        <div className="space-y-4">
          {plateNumbers.map((plateNumber) => (
            <div key={plateNumber}>
              <div className="text-sm font-medium mb-1">
                {reqDisplay} — Plate {plateNumber} of {plateCount}
              </div>
              <PlateGrid plateNumber={plateNumber} interactive={false} />
            </div>
          ))}
        </div>
      </section>
    </div>
  )
}
```
